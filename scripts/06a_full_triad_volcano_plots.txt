## READ COUNT MATRICES

read_count_matrix <- function(file) {

  cat(
    "\nReading:",
    basename(file),
    "\n"
  )

  df <- read_excel(
    file
  )

  if (ncol(df) < 2) {

    stop(
      "Count matrix contains fewer than two columns:\n",
      file
    )

  }

  colnames(df)[1] <- "EnsemblID"

  df$EnsemblID <- as.character(
    df$EnsemblID
  )

## Remove Ensembl version numbers
  df$EnsemblID <- sub(
    "\\..*$",
    "",
    df$EnsemblID
  )

## Remove blank gene IDs
  df <- df %>%

    filter(
      !is.na(EnsemblID),
      EnsemblID != ""
    )

## Convert counts to numeric
  df <- df %>%

    mutate(

      across(
        -EnsemblID,
        ~ as.numeric(.x)
      )

    )

  
# If duplicate Ensembl IDs occur within a batch, sum their counts.
  if (anyDuplicated(df$EnsemblID)) {

    warning(
      "Duplicate Ensembl IDs detected in ",
      basename(file),
      ". Counts will be summed."
    )

    df <- df %>%

      group_by(
        EnsemblID
      ) %>%

      summarise(

        across(
          everything(),
          ~ sum(
            .x,
            na.rm = TRUE
          )
        ),

        .groups = "drop"

      )

  }

  return(df)

}

## FUNCTION TO PREPARE RESULTS


prepare_results <- function(
    dds,
    contrast,
    contrast_name
) {

  res <- results(
    dds,
    contrast = contrast,
    alpha = 0.05
  )


  results_df <- as.data.frame(
    res
  ) %>%

    rownames_to_column(
      "EnsemblID"
    ) %>%

    mutate(

      EnsemblID = sub(
        "\\..*$",
        "",
        EnsemblID
      )

    )


  ## Map Ensembl IDs to genes 
  gene_symbols <- AnnotationDbi::mapIds(

    org.Hs.eg.db,

    keys = results_df$EnsemblID,

    keytype = "ENSEMBL",

    column = "SYMBOL",

    multiVals = "first"

  )


  results_df$Gene <- unname(
    gene_symbols[
      results_df$EnsemblID
    ]
  )


  results_df <- results_df %>%

    mutate(

      Contrast = contrast_name,

      padj_plot = ifelse(
        is.na(padj),
        1,
        padj
      ),

      neg_log10_padj = -log10(
        pmax(
          padj_plot,
          .Machine$double.xmin
        )
      ),

      Significant = case_when(

        !is.na(padj) &
          padj < 0.05 &
          log2FoldChange > 0 ~
          "Upregulated",

        !is.na(padj) &
          padj < 0.05 &
          log2FoldChange < 0 ~
          "Downregulated",

        TRUE ~
          "Not significant"

      )

    )


  return(
    results_df
  )

}


## RUN ALL THREE CONTRASTS

summary_table <- lapply(

        res <- results_list[[contrast_name]]
        
        res_valid <- res %>%
            filter(!is.na(padj))
        
        ## Significant genes
        significant <- res_valid %>%
            filter(padj < 0.05)
        
        ## Upregulated genes
        upregulated <- significant %>%
            filter(log2FoldChange > 0)
        
        ## Downregulated genes
        downregulated <- significant %>%
            filter(log2FoldChange < 0)
        
        tibble(
            Contrast = contrast_name,
            Significant = nrow(significant),
            Upregulated = nrow(upregulated),
            Downregulated = nrow(downregulated)
        )
    }
) %>%
    bind_rows()

## VOLCANO PLOT FUNCTION

make_volcano <- function(
    results_df,
    plot_title,
    x_label
) {


  ## Label the five most significant upregulated and five most significant downregulated genes

  label_df <- results_df %>%

    filter(

      !is.na(padj),

      !is.na(log2FoldChange),

      !is.na(Gene),

      Gene != "",

      padj < 0.05

    ) %>%

    group_by(
      Significant
    ) %>%

    arrange(
      padj
    ) %>%

    slice_head(
      n = 5
    ) %>%

    ungroup()


  ggplot(

    results_df,

    aes(
      x = log2FoldChange,
      y = neg_log10_padj
    )

  ) +

    geom_point(

      aes(
        colour = Significant
      ),

      alpha = 0.65,

      size = 1.5

    ) +

    geom_hline(

      yintercept = -log10(0.05),

      linetype = "dashed",

      linewidth = 0.5

    ) +

    geom_vline(

      xintercept = 0,

      linetype = "dashed",

      linewidth = 0.5

    ) +

    geom_text_repel(

      data = label_df,

      aes(
        label = Gene
      ),

      size = 3.5,

      max.overlaps = Inf,

      box.padding = 0.5,

      point.padding = 0.25,

      min.segment.length = 0,

      show.legend = FALSE

    ) +

    scale_colour_manual(

      values = c(

        "Upregulated" = "firebrick2",

        "Downregulated" = "steelblue2",

        "Not significant" = "grey70"

      )

    ) +

    labs(

      title = plot_title,

      subtitle =
        "Seven-patient matched triad cohort",

      x = x_label,

      y =
        expression(
          -log[10](
            "adjusted p-value"
          )
        ),

      colour = NULL

    ) +

    theme_bw() +

    theme(

      plot.title =
        element_text(
          face = "bold",
          size = 14
        ),

      plot.subtitle =
        element_text(
          size = 11
        ),

      axis.title =
        element_text(
          face = "bold"
        ),

      legend.position =
        "bottom",

      panel.grid.minor =
        element_blank()

    )

}

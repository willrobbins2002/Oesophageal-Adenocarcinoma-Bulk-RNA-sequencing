get_deseq_results <- function(
    dds,
    numerator,
    denominator,
    label
) {
  
  res <- results(
    dds,
    contrast = c(
      "TreatmentModality",
      numerator,
      denominator
    )
  )
  
  res_df <- as.data.frame(res) %>%
    rownames_to_column("ensembl_id") %>%
    arrange(padj)
  
  cat(
    label,
    ": ",
    sum(res_df$padj < 0.05, na.rm = TRUE),
    " genes at FDR < 0.05 out of ",
    sum(!is.na(res_df$pvalue)),
    " tested\n",
    sep = ""
  )
  
  res_df
}

de_results <- list()

for (comp in comparisons) {
  
  de_results[[comp$label]] <- get_deseq_results(
    dds = dds_oac_unadjusted,
    numerator = comp$numerator,
    denominator = comp$denominator,
    label = comp$label
  )
}

## RUN PRERANKED GSEA

run_fgsea <- function(
    res_df,
    pathways,
    label,
    minSize = 15,
    maxSize = 500
) {
  
  required_cols <- c(
    "ensembl_id",
    "gene_symbol",
    "stat"
  )
  
  missing_cols <- setdiff(
    required_cols,
    colnames(res_df)
  )
  
  if (length(missing_cols) > 0) {
    stop(
      "DESeq2 results are missing required columns: ",
      paste(missing_cols, collapse = ", ")
    )
  }
  
  rank_df <- res_df %>%
    filter(
      !is.na(gene_symbol),
      gene_symbol != "",
      !is.na(stat),
      is.finite(stat)
    ) %>%
    group_by(gene_symbol) %>%
    slice_max(
      order_by = abs(stat),
      n = 1,
      with_ties = FALSE
    ) %>%
    ungroup() %>%
    select(gene_symbol, stat)
  
  ranks <- deframe(rank_df)
  
  ranks <- sort(
    ranks,
    decreasing = TRUE
  )
  
  cat("\n", label, "\n", sep = "")
  cat("Ranked genes:", length(ranks), "\n")
  cat("Positive Wald statistics:", sum(ranks > 0), "\n")
  cat("Negative Wald statistics:", sum(ranks < 0), "\n")
  
  if (!any(ranks > 0) || !any(ranks < 0)) {
    stop(
      "The GSEA ranking has only one direction. ",
      "Use signed DESeq2 `stat` values without filtering to DEGs."
    )
  }
  
  pathway_genes <- unique(
    unlist(pathways)
  )
  
  overlap_n <- sum(
    names(ranks) %in% pathway_genes
  )
  
  cat(
    "Genes overlapping Hallmark pathways:",
    overlap_n,
    "\n"
  )
  
  if (overlap_n < 100) {
    warning(
      "Low overlap between ranked gene symbols and pathway gene symbols. ",
      "Check gene identifier mapping."
    )
  }
  
  fgseaMultilevel(
    pathways = pathways,
    stats = ranks,
    minSize = minSize,
    maxSize = maxSize,
    scoreType = "std"
  ) %>%
    as.data.frame() %>%
    arrange(padj)
}

    sep = ""
  )
}


## CREATE BALANCED TOP-10 GSEA PATHWAY PLOTS
## Five positive-NES pathways: enriched in the first-named group.
## Five negative-NES pathways: enriched in the second-named group.

make_top10_gsea_plot <- function(
    fg_res,
    label,
    group_1,
    group_2,
    n_total = 10
) {
  
  n_each <- floor(
    n_total / 2
  )
  
  plot_input <- fg_res %>%
    filter(
      !is.na(NES),
      !is.na(padj),
      is.finite(NES),
      is.finite(padj)
    )
  
  if (nrow(plot_input) == 0) {
    stop(
      label,
      ": no valid GSEA pathway results were returned."
    )
  }
  
  top_positive <- plot_input %>%
    filter(NES > 0) %>%
    arrange(
      padj,
      desc(NES)
    ) %>%
    slice_head(n = n_each)
  
  top_negative <- plot_input %>%
    filter(NES < 0) %>%
    arrange(
      padj,
      NES
    ) %>%
    slice_head(n = n_each)
  
  plot_df <- bind_rows(
    top_positive,
    top_negative
  ) %>%
    mutate(
      direction = if_else(
        NES > 0,
        paste0("Enriched in ", group_1),
        paste0("Enriched in ", group_2)
      ),
      pathway_label = pathway %>%
        str_remove("^HALLMARK_") %>%
        str_replace_all("_", " ") %>%
        str_to_title()
    ) %>%
    arrange(NES) %>%
    mutate(
      pathway_label = factor(
        pathway_label,
        levels = unique(pathway_label)
      )
    )
  
  if (nrow(plot_df) == 0) {
    stop(
      label,
      ": no positive or negative NES pathways were available."
    )
  }
  
  fill_colours <- setNames(
    c(
      "#C0392B",
      "#2874A6"
    ),
    c(
      paste0("Enriched in ", group_1),
      paste0("Enriched in ", group_2)
    )
  )
  
  ggplot(
    plot_df,
    aes(
      x = pathway_label,
      y = NES,
      fill = direction
    )
  ) +
    geom_col(
      width = 0.75
    ) +
    geom_hline(
      yintercept = 0,
      linewidth = 0.4,
      colour = "black"
    ) +
    coord_flip() +
    scale_fill_manual(
      values = fill_colours
    ) +
    labs(
      title = paste0(
        "Hallmark GSEA: ",
        label
      ),
      subtitle = paste0(
        "Top ",
        n_total,
        " pathways: ",
        n_each,
        " enriched in each group, ranked by adjusted p-value"
      ),
      x = NULL,
      y = "Normalised enrichment score (NES)",
      fill = NULL
    ) +
    theme_classic(
      base_size = 12
    ) +
    theme(
      legend.position = "bottom",
      axis.text.y = element_text(size = 9),
      plot.title = element_text(face = "bold")
    )
}
